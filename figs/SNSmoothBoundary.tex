\begin{tikzpicture}[
	baseline={(current bounding box.center)},
	scale=.65
	]
	
	\def\l{1.3}
	\def\c{cos(60)}
	\def\s{sin(60)}
	\def\Nx{3}
	\def\Ny{3}
	
	\clip ({-(\c+1.5)*\l},{-1.5*\s*\l}) rectangle+ ({2*(\Nx+1)*\l*(1+\c) - \l/2},{(2*\Ny+1)*\l*\s});
	
	% Unit cell A: centered at (2y*l*(1+s), -2cx*l)
	\newcommand{\cellA}[2]{
		\begin{scope}[obj1, shift={({2*\l*#1*(1+\c)},{2*\s*\l*#2})}]
			\draw (0,0) --+ (120:\l/2);
			\draw (0,0) --+ (240:\l/2);
			\draw (0,0) -- (\l,0);
			\draw (\l,0) --+ (60:\l/2);
			\draw (\l,0) --+ (-60:\l/2);
			\node[mor] at (0,0) {};
			\node[mor] at (\l,0) {};
		\end{scope}
	}
	
	% Unit cell B: shifted by (l*(1+s), (2x-1)*c*l) ... rotated
	\newcommand{\cellB}[2]{
		\begin{scope}[obj1, shift={({2*\l*#1*(1+\c)-\l*(1+\c)},{(2*#2-1)*\s*\l})}]
			\draw (0,0) --+ (120:\l/2);
			\draw (0,0) --+ (240:\l/2);
			\draw (0,0) -- (\l,0);
			\draw (\l,0) --+ (60:\l/2);
			\draw (\l,0) --+ (-60:\l/2);
			\node[mor] at (0,0) {};
			\node[mor] at (\l,0) {};
		\end{scope}
	}
	
	% Bottom row (y=0): A cells + special B cells with obj3 vertical bar
	\foreach \y in {0,...,\Ny} {
		\cellA{0}{\y}
		\begin{scope}[obj1, shift={({-\l*(1+\c)},{(2*\y-1)*\s*\l})}]
			\draw[obj3] (0,{-.5*\s*\l}) -- (0,{1.5*\s*\l});
			\draw (0,0) -- (\l,0);
			\draw (\l,0) --+ (60:\l/2);
			\draw (\l,0) --+ (-60:\l/2);
			\node[mor] at (0,0) {};
			\node[mor] at (\l,0) {};
		\end{scope}
	}
	
	\foreach \y in {0,...,\Ny}
	\foreach \x in {1,...,\Nx} {
		\cellA{\x}{\y}
		\cellB{\x}{\y}
	}
	
\end{tikzpicture}
