\begin{tikzpicture}[
	baseline={(current bounding box.center)},
	scale=.65
	]
	
	\def\l{1.3}
	\def\c{cos(30)}
	\def\s{sin(30)}
	\def\Nx{5}
	\def\Ny{2}
	
	\clip ({.5*\c*\l},{.5*\l}) rectangle ({(2*\Nx+.5)*\c*\l},{(2*\Ny+1)*(\l+\s*\l) + .5*\l});
	
	% Unit cell A: centered at (2y*l*(1+s), -2cx*l)
	\newcommand{\cellA}[2]{
		\begin{scope}[obj1, shift={({2*\l*\c*#1},{2*(\s+1)*\l*#2})}]
			\draw (0,0) --+ (210:\l/2);
			\draw (0,0) --+ (-30:\l/2);
			\draw (0,0) -- (0,\l);
			\draw (0,\l) --+ (150:\l/2);
			\draw (0,\l) --+ (30:\l/2);
			\node[mor] at (0,0) {};
			\node[mor] at (0,\l) {};
		\end{scope}
	}
	
	% Unit cell B: shifted by (l*(1+s), (2x-1)*c*l) ... rotated
	\newcommand{\cellB}[2]{
		\begin{scope}[obj1, shift={({\l*\c*(2*#1+1)},{2*(\s+1)*\l*#2 + \l + \s*\l})}]
			\draw (0,0) --+ (210:\l/2);
			\draw (0,0) --+ (-30:\l/2);
			\draw (0,0) -- (0,\l);
			\draw (0,\l) --+ (150:\l/2);
			\draw (0,\l) --+ (30:\l/2);
			\node[mor] at (0,0) {};
			\node[mor] at (0,\l) {};
		\end{scope}
	}
	
	\foreach \y in {0,...,\Ny}
	\foreach \x in {0,...,\Nx} {
		\cellA{\x}{\y}
		\cellB{\x}{\y}
	}
	
	% Reference points
	\node[fill=white] at ({6*\c*\l},{4*\s*\l + 4.5*\l}) {};
	\node[fill=white] at ({5.5*\c*\l},{3.5*\s*\l + 4*\l}) {};
	\node[fill=white] at ({5*\c*\l},{3*\s*\l + 3.5*\l}) {};
	\node[fill=white] at ({3.5*\c*\l},{2.5*\s*\l + 3*\l}) {};
	\node[fill=white] at ({4*\c*\l},{2*\s*\l + 2.5*\l}) {};

	\draw[obj2, wiggly] ({7*\c*\l},{4*\s*\l + 4.5*\l}) -- ({5*\c*\l},{4*\s*\l + 4.5*\l});
	\draw[obj2, wiggly] ({5*\c*\l},{4*\s*\l + 4.5*\l}) -- ({6*\c*\l},{3*\s*\l + 3.5*\l});
	\draw[obj2, wiggly] ({6*\c*\l},{3*\s*\l + 3.5*\l}) -- ({4*\c*\l},{3*\s*\l + 3.5*\l});
	\draw[obj2, wiggly] ({4*\c*\l},{3*\s*\l + 3.5*\l}) -- ({3*\c*\l},{2*\s*\l + 2.5*\l});
	\draw[obj2, wiggly] ({3*\c*\l},{2*\s*\l + 2.5*\l}) -- ({5*\c*\l},{2*\s*\l + 2.5*\l});
	
	\node[above] at ({4*\c*\l},{3*\s*\l + 3.5*\l}) {$\sss Z$};
\end{tikzpicture}
